-- 1. Create and Select the Database (Matching your views.py)
CREATE DATABASE IF NOT EXISTS nodue_new;
USE nodue_new;

-- 2. Create Employees Table
CREATE TABLE employees (
    emp_name VARCHAR(40),
    gender VARCHAR(20),
    contact_no VARCHAR(20),
    email VARCHAR(50), 
    qualification VARCHAR(65),
    experience VARCHAR(45),
    user_role VARCHAR(35),
    username VARCHAR(50) PRIMARY KEY,
    password VARCHAR(50)
);

-- 3. Create Student Table
CREATE TABLE student (
    student_id VARCHAR(50),
    student_name VARCHAR(40),
    gender VARCHAR(20),
    contact_no VARCHAR(20),
    email VARCHAR(50), 
    address VARCHAR(200),
    course VARCHAR(50),
    course_year VARCHAR(20),
    password VARCHAR(50),
    library_fee DOUBLE,
    hostel_fee DOUBLE,
    tution_fee DOUBLE,
    exam_fee DOUBLE,
    lab_hod_fee DOUBLE,
    PRIMARY KEY(student_id, course_year)
);

-- 4. Create Payments Table
CREATE TABLE payments (
    student_id VARCHAR(50),
    course_year VARCHAR(20),
    paying_dept VARCHAR(50),
    amount DOUBLE,
    payment_date VARCHAR(30)
);

-- 5. Create Document Submissions Table (For AI Feature)
CREATE TABLE document_submissions (
    id INT AUTO_INCREMENT PRIMARY KEY,
    student_id VARCHAR(50),
    department VARCHAR(50),
    document_type VARCHAR(50),
    file_path VARCHAR(255),
    uploaded_at DATETIME DEFAULT CURRENT_TIMESTAMP,
    ai_verification_status VARCHAR(20),
    ai_confidence_score DOUBLE
);
